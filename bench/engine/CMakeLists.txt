cmake_minimum_required(VERSION 3.22)
project(tebench VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release)
set(TE_ADD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(DAWD ${CMAKE_CURRENT_SOURCE_DIR}/../../dawd/source)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../tracktion_engine tracktion_build)
juce_add_console_app(tebench NEEDS_CURL FALSE NEEDS_WEB_BROWSER FALSE)
target_compile_definitions(tebench PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_ALSA=1 JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_LV2=1)
target_compile_features(tebench PRIVATE cxx_std_20)
target_include_directories(tebench PRIVATE ${DAWD})
target_sources(tebench PRIVATE main.cpp ${DAWD}/Routing.cpp ${DAWD}/Feedback.cpp)
target_link_libraries(tebench PRIVATE tracktion::tracktion_engine "-latomic")
